cmake_minimum_required(VERSION 3.16)
project(fpdb_tide LANGUAGES CXX)

add_library(fpdb-tide src/tide.cpp src/scheduler.cpp)
target_include_directories(fpdb-tide PUBLIC include)
target_compile_features(fpdb-tide PUBLIC cxx_std_17)
target_compile_options(fpdb-tide PRIVATE -Wall -Wextra -Wpedantic)

option(TIDE_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(TIDE_ENABLE_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(fpdb-tide PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fpdb-tide PUBLIC -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "Tide sanitizers require Clang or GCC")
  endif()
endif()

add_executable(tide-adapter-demo examples/database_adapter.cpp)
target_link_libraries(tide-adapter-demo PRIVATE fpdb-tide)

option(TIDE_BUILD_TESTS "Build Tide unit tests" ON)
if(TIDE_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(tide-consumer-test tests/consumer_test.cpp)
  target_link_libraries(tide-consumer-test PRIVATE fpdb-tide Threads::Threads)
  add_test(NAME tide-consumer-test COMMAND tide-consumer-test)
  add_executable(tide-test tests/tide_test.cpp)
  target_link_libraries(tide-test PRIVATE fpdb-tide)
  add_test(NAME tide-test COMMAND tide-test)
  add_executable(tide-scheduler-test tests/scheduler_test.cpp)
  target_link_libraries(tide-scheduler-test PRIVATE fpdb-tide)
  add_test(NAME tide-scheduler-test COMMAND tide-scheduler-test)
  add_executable(tide-lifecycle-test tests/lifecycle_test.cpp)
  target_link_libraries(tide-lifecycle-test PRIVATE fpdb-tide Threads::Threads)
  add_test(NAME tide-lifecycle-test COMMAND tide-lifecycle-test)
  add_executable(tide-shared-memory-test tests/shared_memory_test.cpp)
  target_link_libraries(tide-shared-memory-test PRIVATE fpdb-tide Threads::Threads)
  add_test(NAME tide-shared-memory-test COMMAND tide-shared-memory-test)
  add_executable(tide-adapter-contract-test tests/adapter_contract_test.cpp)
  target_link_libraries(tide-adapter-contract-test PRIVATE fpdb-tide)
  add_test(NAME tide-adapter-contract-test COMMAND tide-adapter-contract-test)
  add_test(NAME tide-adapter-demo COMMAND tide-adapter-demo)
  add_executable(tide-recovery-test tests/recovery_test.cpp)
  target_link_libraries(tide-recovery-test PRIVATE fpdb-tide)
  add_test(NAME tide-recovery-test COMMAND tide-recovery-test)
  add_executable(tide-recovery-bench tests/recovery_bench.cpp)
  target_link_libraries(tide-recovery-bench PRIVATE fpdb-tide)
  # Existing fixtures use assert for both checks and operations. Keep those
  # expressions active in optimized builds as well.
  foreach(test_target tide-test tide-recovery-test tide-recovery-bench)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
